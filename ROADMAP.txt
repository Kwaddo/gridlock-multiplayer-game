### 1B. Crosshair preset data model

**New file:** `client/src/weapons/crosshair_presets.rs`
**Register in:** `client/src/weapons/mod.rs` — add `pub mod crosshair_presets;`

```rust
use serde::{Serialize, Deserialize};
use std::path::PathBuf;
use super::crosshair::{CrosshairConfig, CrosshairType};
use super::stats::WeaponKind;
use macroquad::prelude::Color;

#[derive(Clone, Serialize, Deserialize)]
pub struct CrosshairPreset {
    pub name: String,
    pub is_default: bool,           // true = factory preset, can't overwrite/delete
    pub crosshair_type: CrosshairType,
    pub color: [u8; 4],             // RGBA (Color doesn't impl Serialize)
    pub dot_enabled: bool,
    pub dot_size: f32,
    pub line_length: f32,
    pub line_thickness: f32,
    pub center_gap: f32,
    pub outline_enabled: bool,
    pub outline_thickness: f32,
    pub kick_per_shot: f32,
    pub kick_decay_rate: f32,
    pub kick_max: f32,
}

impl CrosshairPreset {
    /// Convert from runtime CrosshairConfig to serializable preset
    pub fn from_config(name: &str, config: &CrosshairConfig, is_default: bool) -> Self {
        Self {
            name: name.to_string(),
            is_default,
            crosshair_type: config.crosshair_type,
            color: [
                (config.color.r * 255.0) as u8,
                (config.color.g * 255.0) as u8,
                (config.color.b * 255.0) as u8,
                (config.color.a * 255.0) as u8,
            ],
            dot_enabled: config.dot_enabled,
            dot_size: config.dot_size,
            line_length: config.line_length,
            line_thickness: config.line_thickness,
            center_gap: config.center_gap,
            outline_enabled: config.outline_enabled,
            outline_thickness: config.outline_thickness,
            kick_per_shot: config.kick_per_shot,
            kick_decay_rate: config.kick_decay_rate,
            kick_max: config.kick_max,
        }
    }

    /// Apply this preset to a CrosshairConfig
    pub fn apply_to(&self, config: &mut CrosshairConfig) {
        config.crosshair_type = self.crosshair_type;
        config.color = Color::from_rgba(self.color[0], self.color[1], self.color[2], self.color[3]);
        config.dot_enabled = self.dot_enabled;
        config.dot_size = self.dot_size;
        config.line_length = self.line_length;
        config.line_thickness = self.line_thickness;
        config.center_gap = self.center_gap;
        config.outline_enabled = self.outline_enabled;
        config.outline_thickness = self.outline_thickness;
        config.kick_per_shot = self.kick_per_shot;
        config.kick_decay_rate = self.kick_decay_rate;
        config.kick_max = self.kick_max;
    }
}

pub struct CrosshairPresetManager {
    pub presets: Vec<CrosshairPreset>,
    file_path: PathBuf,
}

impl CrosshairPresetManager {
    pub fn new() -> Self {
        let mut mgr = Self {
            presets: Vec::new(),
            file_path: PathBuf::from("crosshair_presets.json"),
        };
        mgr.rebuild_defaults();
        mgr.load_user_presets();
        mgr
    }

    /// Regenerate factory defaults (always first in list)
    fn rebuild_defaults(&mut self) {
        self.presets.clear();
        for kind in WeaponKind::ALL {
            let config = CrosshairConfig::for_weapon(kind);
            let name = format!("{} Default", kind.stats().name);
            self.presets.push(CrosshairPreset::from_config(&name, &config, true));
        }
    }

    /// Load user presets from JSON and append after defaults
    fn load_user_presets(&mut self) {
        if let Ok(data) = std::fs::read_to_string(&self.file_path) {
            if let Ok(user_presets) = serde_json::from_str::<Vec<CrosshairPreset>>(&data) {
                for mut p in user_presets {
                    p.is_default = false; // safety: never trust file marking itself as default
                    self.presets.push(p);
                }
            }
        }
    }

    /// Save only user (non-default) presets to disk
    pub fn save(&self) {
        let user_only: Vec<&CrosshairPreset> = self.presets.iter()
            .filter(|p| !p.is_default)
            .collect();
        if let Ok(json) = serde_json::to_string_pretty(&user_only) {
            let _ = std::fs::write(&self.file_path, json);
        }
    }

    /// Create a new user preset from current config
    pub fn save_as_new(&mut self, name: String, config: &CrosshairConfig) {
        self.presets.push(CrosshairPreset::from_config(&name, config, false));
        self.save();
    }

    /// Overwrite an existing user preset (panics if default)
    pub fn overwrite(&mut self, index: usize, config: &CrosshairConfig) {
        if index < self.presets.len() && !self.presets[index].is_default {
            let name = self.presets[index].name.clone();
            self.presets[index] = CrosshairPreset::from_config(&name, config, false);
            self.save();
        }
    }

    /// Delete a user preset (no-op if default)
    pub fn delete(&mut self, index: usize) {
        if index < self.presets.len() && !self.presets[index].is_default {
            self.presets.remove(index);
            self.save();
        }
    }

    pub fn default_count(&self) -> usize {
        self.presets.iter().filter(|p| p.is_default).count()
    }

    /// Suggested name for next custom preset
    pub fn next_custom_name(&self) -> String {
        let custom_count = self.presets.iter().filter(|p| !p.is_default).count();
        format!("Custom {}", custom_count + 1)
    }
}
```

### 1F. Text input widget

**Hint:** macroquad has `get_char_pressed()` and `is_key_pressed(KeyCode::Backspace)`.

```rust
fn update_text_input(buffer: &mut String, max_len: usize) -> Option<bool> {
    // Returns Some(true) on Enter, Some(false) on Escape, None if still editing
    if is_key_pressed(KeyCode::Enter) { return Some(true); }
    if is_key_pressed(KeyCode::Escape) { return Some(false); }

    if is_key_pressed(KeyCode::Backspace) && !buffer.is_empty() {
        buffer.pop();
    }

    while let Some(ch) = get_char_pressed() {
        if ch.is_alphanumeric() || ch == ' ' || ch == '-' || ch == '_' {
            if buffer.len() < max_len {
                buffer.push(ch);
            }
        }
    }
    None
}

fn draw_text_input(buffer: &str, x: f32, y: f32, w: f32) {
    draw_rectangle(x, y, w, 30.0, Color::from_rgba(20, 20, 20, 255));
    draw_rectangle_lines(x, y, w, 30.0, 2.0, Color::from_rgba(100, 180, 255, 255));

    let display = if buffer.is_empty() { "Type name..." } else { buffer };
    let color = if buffer.is_empty() { GRAY } else { WHITE };
    draw_text_ex(display, x + 8.0, y + 22.0, text_params(18, color));

    // blinking cursor
    let cursor_x = x + 8.0 + measure_game_text(buffer, 18);
    if (get_time() * 2.0) as i32 % 2 == 0 {
        draw_line(cursor_x, y + 5.0, cursor_x, y + 25.0, 1.5, WHITE);
    }
}
```

### 2A. Dev Mode: Aim Trace Line

**Files:** `client/src/screens/playing/rendering.rs`, `client/src/screens/playing.rs`

When `show_hitboxes` is ON, draw a continuous 3D line from player's eye to aim hit point.
Color-coded: head=red, torso=green, limbs=blue, miss=dim white.

```rust
// In rendering.rs, add to impl PlayingState:
pub(super) fn draw_aim_trace(&self) {
    use crate::weapons::combat::CombatEntity;

    let eye = self.position + vec3(0.0, 1.6, 0.0);
    let fwd = vec3(
        self.yaw.cos() * self.pitch.cos(),
        self.pitch.sin(),
        self.yaw.sin() * self.pitch.cos(),
    );

    let stats = self.weapons.active().stats();
    let range = stats.bullet_range;

    let mut closest: Option<(f32, crate::weapons::combat::HitZone)> = None;

    if let Some((zone, dist)) = self.target_dummy.ray_test(eye, fwd, range) {
        closest = Some((dist, zone));
    }
    for p in &self.other_players {
        if let Some((zone, dist)) = p.ray_test(eye, fwd, range) {
            if closest.map_or(true, |(d, _)| dist < d) {
                closest = Some((dist, zone));
            }
        }
    }

    let (end_point, color) = if let Some((dist, zone)) = closest {
        let c = match zone {
            crate::weapons::combat::HitZone::Head => Color::from_rgba(255, 50, 50, 200),
            crate::weapons::combat::HitZone::Torso => Color::from_rgba(50, 255, 50, 200),
            crate::weapons::combat::HitZone::Limbs => Color::from_rgba(50, 150, 255, 200),
        };
        (eye + fwd * dist, c)
    } else {
        (eye + fwd * range, Color::from_rgba(255, 255, 255, 40))
    };

    let up = vec3(0.0, 1.0, 0.0);
    let side = fwd.cross(up).normalize() * 0.01;
    draw_line_3d(eye, end_point, color);
    draw_line_3d(eye + side, end_point + side, color);
    draw_line_3d(eye - side, end_point - side, color);

    if closest.is_some() {
        draw_sphere(end_point, 0.05, None, color);
    }
}
```

In `playing.rs`, after `self.draw_tracers();`:

```rust
    self.draw_aim_trace();
}
```

### 5B. Sliding

**File:** `client/src/physics/movement.rs`

```rust
// Sprint + crouch while moving fast → slide
pub is_sliding: bool,
pub slide_timer: f32,
pub slide_velocity: Vec3,

// Trigger: if sprinting && crouch pressed && speed > threshold
// Duration: ~0.8 seconds
// Camera height: crouch height
// Speed: starts at sprint speed * 1.3, decays to 0
// Can't shoot during slide (or hipfire only, no ADS)

if is_sprinting && crouch_just_pressed && horizontal_speed > 4.0 {
    is_sliding = true;
    slide_timer = 0.8;
    slide_velocity = forward * horizontal_speed * 1.3;
}

if is_sliding {
    slide_timer -= delta;
    slide_velocity *= 0.95; // friction decay
    position += slide_velocity * delta;
    if slide_timer <= 0.0 { is_sliding = false; }
}
```

- `Mesh { vertices: Vec<Vertex>, indices: Vec<u16>, texture: Option<Texture2D> }`
- `draw_mesh(&mesh)` renders custom geometry
- No native OBJ/GLTF loader — use `tobj` crate to parse OBJ files

**Concrete pipeline:**
1. Source Minecraft resource pack (blocky aesthetic matches game)
2. Export weapon models as OBJ (many free tools: Blockbench, MCreator)
3. Use `tobj` crate to load OBJ at runtime: `tobj::load_obj("gun.obj", &options)`
4. Convert tobj vertices → macroquad `Vertex` + `Mesh`
5. Load texture with `load_texture("gun_texture.png")`
6. Render with `draw_mesh()` in first-person viewport

**Cargo.toml addition when ready:**
```toml
tobj = "4"
```

**Why not Bevy?**
- Bevy and macroquad CANNOT run together (both own the window + GPU context)
- Bevy gives you PBR, shadows, GLTF loading — but its render graph is complex and undocumented
- You lose all the pixel-level control you have now (crosshairs, scope, HUD)
- macroquad + tobj gives you model loading while keeping full control
- The blocky Minecraft aesthetic means you don't NEED PBR or fancy shaders

**Weapon switch animation** (works even without 3D models):

```rust
// Use switch_cooldown timer on WeaponInventory
// Fade crosshair out, bob HUD weapon icon down, swap, bob up, fade in
let switch_progress = 1.0 - (weapon_inventory.switch_cooldown / SWITCH_TIME);
let bob_y = if switch_progress < 0.5 {
    switch_progress * 2.0 * 30.0 // lowering: 0 → 30px down
} else {
    (1.0 - (switch_progress - 0.5) * 2.0) * 30.0 // raising: 30px → 0
};
```
