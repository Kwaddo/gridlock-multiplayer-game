import bpy

def find_surface_shader(output_node):
    if not output_node or not output_node.inputs["Surface"].is_linked:
        return None
    return output_node.inputs["Surface"].links[0].from_node

def find_material_output(nodes):
    for n in nodes:
        if n.type == "OUTPUT_MATERIAL" and n.is_active_output:
            return n
    for n in nodes:
        if n.type == "OUTPUT_MATERIAL":
            return n
    return None

def link_or_copy(links, src_input, dst_input):
    if src_input is None or dst_input is None:
        return
    if src_input.is_linked:
        links.new(src_input.links[0].from_socket, dst_input)
    else:
        try:
            dst_input.default_value = src_input.default_value
        except Exception:
            pass

def first_input(node, names):
    if not node:
        return None
    lowered = {n.lower() for n in names}
    for inp in node.inputs:
        if inp.name.lower() in lowered:
            return inp
    return None

def ensure_normal_map_chain(nodes, links, src_input, principled):
    # If source has a normal map chain, preserve it.
    if not src_input or not src_input.is_linked:
        return False

    from_socket = src_input.links[0].from_socket
    from_node = from_socket.node

    # If already a Normal Map node, use as is.
    if from_node.type == "NORMAL_MAP":
        links.new(from_socket, principled.inputs["Normal"])
        return True

    # If it's an image texture or anything else directly, insert Normal Map node.
    normal_map = nodes.new("ShaderNodeNormalMap")
    normal_map.location = (principled.location.x - 220, principled.location.y - 260)
    links.new(from_socket, normal_map.inputs["Color"])
    links.new(normal_map.outputs["Normal"], principled.inputs["Normal"])
    return True

converted = 0
skipped = 0

for mat in bpy.data.materials:
    if not mat.use_nodes or not mat.node_tree:
        skipped += 1
        continue

    nodes = mat.node_tree.nodes
    links = mat.node_tree.links

    output = find_material_output(nodes)
    if not output:
        skipped += 1
        continue

    shader = find_surface_shader(output)

    # If already principled, keep it.
    if shader and shader.type == "BSDF_PRINCIPLED":
        skipped += 1
        continue

    # Create principled shader
    principled = nodes.new("ShaderNodeBsdfPrincipled")
    principled.location = (output.location.x - 320, output.location.y)

    # Try to map common inputs from old shader
    base_color_in = first_input(shader, ["Base Color", "Diffuse", "Color"])
    roughness_in = first_input(shader, ["Roughness", "Glossiness"])
    metallic_in = first_input(shader, ["Metallic", "Specular"])
    alpha_in = first_input(shader, ["Alpha", "Opacity"])
    emission_color_in = first_input(shader, ["Emission", "Emission Color"])
    emission_strength_in = first_input(shader, ["Emission Strength"])
    normal_in = first_input(shader, ["Normal", "Normal Map"])

    # Base color
    link_or_copy(links, base_color_in, principled.inputs["Base Color"])

    # Roughness / Glossiness handling:
    # If source has "Glossiness", invert to roughness when constant.
    if roughness_in:
        if roughness_in.name.lower() == "glossiness" and not roughness_in.is_linked:
            try:
                principled.inputs["Roughness"].default_value = 1.0 - float(roughness_in.default_value)
            except Exception:
                pass
        elif roughness_in.is_linked and roughness_in.name.lower() == "glossiness":
            invert = nodes.new("ShaderNodeInvert")
            invert.location = (principled.location.x - 220, principled.location.y - 120)
            links.new(roughness_in.links[0].from_socket, invert.inputs["Color"])
            links.new(invert.outputs["Color"], principled.inputs["Roughness"])
        else:
            link_or_copy(links, roughness_in, principled.inputs["Roughness"])

    # Metallic (if mapped from old "Specular", keep subtle)
    if metallic_in:
        if metallic_in.name.lower() == "specular" and not metallic_in.is_linked:
            try:
                v = float(metallic_in.default_value)
                principled.inputs["Metallic"].default_value = max(0.0, min(1.0, v * 0.2))
            except Exception:
                pass
        else:
            link_or_copy(links, metallic_in, principled.inputs["Metallic"])

    # Alpha
    if alpha_in:
        link_or_copy(links, alpha_in, principled.inputs["Alpha"])
        mat.blend_method = "BLEND"

    # Emission
    if emission_color_in:
        link_or_copy(links, emission_color_in, principled.inputs["Emission Color"])
    if emission_strength_in:
        link_or_copy(links, emission_strength_in, principled.inputs["Emission Strength"])

    # Normal
    if normal_in:
        ensure_normal_map_chain(nodes, links, normal_in, principled)

    # Ensure sane defaults if untouched
    if not principled.inputs["Base Color"].is_linked:
        c = principled.inputs["Base Color"].default_value
        # If near-black, lift it slightly so assets don't export almost black by accident.
        if c[0] < 0.03 and c[1] < 0.03 and c[2] < 0.03:
            principled.inputs["Base Color"].default_value = (0.5, 0.5, 0.5, 1.0)

    # Connect principled to output
    links.new(principled.outputs["BSDF"], output.inputs["Surface"])

    # Remove old shader node
    if shader:
        nodes.remove(shader)

    converted += 1

print(f"Converted: {converted}, skipped: {skipped}")
